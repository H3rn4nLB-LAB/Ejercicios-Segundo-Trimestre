DTD - Ejercicio 1 - Elementos sin atributos - Corregir errores en los elementos

Los siguientes documentos no son válidos porque contienen uno o dos errores (los errores no están en la DTD interna). 
Corrija los errores y compruebe con XML Copy Editor que ya son documentos válidos.

DTD - Ejercicio 1-1 - Números

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE numeros [
  <!ELEMENT numeros (#PCDATA)>
]>

<numeros>
  <numero>25</numero>
</numeros>

ERROR: el elemento <numeros> es #PCDATA (texto), pero contiene un elemento <numero> no definido en la DTD.

Corregido: se elimina el elemento <numero> y el #PCDATA

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE numeros [
  <!ELEMENT numeros (#PCDATA)>
]>

<numeros>
 25
</numeros>


DTD - Ejercicio 1-2 - Letras

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE letras [
  <!ELEMENT letras (letra)>
  <!ELEMENT letra (#PCDATA)>
]>

<letras>
  <letra>m</letra>
  <letra>uve doble</letra>
</letras>

ERROR: el elemento <letras> debería contener un único elemento <letra>, pero contiene dos.

Corregido: se elimina alguno de los dos elementos <letra>

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE letras [
  <!ELEMENT letras (letra)>
  <!ELEMENT letra (#PCDATA)>
]>

<letras>
  <letra>m</letra>
</letras>


DTD - Ejercicio 1-3 - Colores

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE colores [
  <!ELEMENT colores (color*)>
  <!ELEMENT color (#PCDATA)>
]>

<colores>
  <color>azul marino</color>
  negro
  <color>amarillo</color>
</colores>

ERROR: el elemento <colores> debería contener únicamente elementos <color>, pero no texto sin etiquetas

Corregido:  se podrían añadir etiquetas <color> alrededor del texto sin etiqueta.

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE colores [
  <!ELEMENT colores (color*)>
  <!ELEMENT color (#PCDATA)>
]>

<colores>
  <color>azul marino</color>
  <color>negro</color>
  <color>amarillo</color>
</colores>
  

DTD - Ejercicio 1-4 - Flores
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE flores [
  <!ELEMENT flores (flor+)>
  <!ELEMENT flor (#PCDATA)>
]>

<flores>
</flores>

ERROR: El elemento <flores> debería contener al menos un elemento <flor>, pero no tiene ninguno.

Corregido: habría que añadir alguna etiqueta <flor>.


<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE flores [
  <!ELEMENT flores (flor+)>
  <!ELEMENT flor (#PCDATA)>
]>

<flores>
  <flor>Rosa</flor>
</flores>
  
  
DTD - Ejercicio 1-5 - Animales
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE animales [
  <!ELEMENT animales (animal*)>
  <!ELEMENT animal (#PCDATA)>
]>

<animales>
  <perro>Caniche</perro>
  <gato>Siamés</gato>
</animales>

ERROR: El elemento <animales> sólo puede contener elementos <animal>, pero los elementos son <perro> y <gato>, elementos que no están definidos.

Corregido: se cambia los elementos por <animales>


<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE animales [
  <!ELEMENT animales (animal*)>
  <!ELEMENT animal (#PCDATA)>
]>

<animales>
  <animal> Perro Caniche </animal>
  <animal> Gato Siamés </animal>
</animales>
  
  
DTD - Ejercicio 1-6 - Escritores
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE escritores [
  <!ELEMENT escritores (escritor*)>
  <!ELEMENT escritor (nombre, nacimiento)>
  <!ELEMENT nombre (#PCDATA)>
  <!ELEMENT nacimiento (#PCDATA)>
]>

<escritores>
  <escritor>
    <nombre>Mario Vargas Llosa</nombre>
    <nacimiento>28 de marzo de 1936</nacimiento>
  </escritor>
  <escritor>
    <nacimiento>1 de abril de 1929</nacimiento>
    <nombre>Milan Kundera</nombre>
  </escritor>
</escritores>

ERROR: el segundo elemento <escritor> debe contener los elementos <nombre> y <nacimiento> en ese orden, pero los tiene al revés.

Corregido: se pone los elementos <nombre> y <nacimiento> ordenados correctamente.
 
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE escritores [
  <!ELEMENT escritores (escritor*)>
  <!ELEMENT escritor (nombre, nacimiento)>
  <!ELEMENT nombre (#PCDATA)>
  <!ELEMENT nacimiento (#PCDATA)>
]>

<escritores>
  <escritor>
    <nombre>Mario Vargas Llosa</nombre>
    <nacimiento>28 de marzo de 1936</nacimiento>
  </escritor>
  <escritor>
    <nombre>Milan Kundera</nombre>
    <nacimiento>1 de abril de 1929</nacimiento>
  </escritor>
</escritores>

  
DTD - Ejercicio 1-7 - Músicos
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE musicos [
  <!ELEMENT musicos (musico*)>
  <!ELEMENT musico ((nombre | apodo), fechaNacimiento)>
  <!ELEMENT nombre (#PCDATA)>
  <!ELEMENT apodo (#PCDATA)>
  <!ELEMENT fechaNacimiento (#PCDATA)>
]>

<musicos>
  <musico>
    <nombre>Antonio Vivaldi</nombre>
    <apodo>El cura pelirrojillo</apodo>
    <fechaNacimiento>4 de marzo de 1678</fechaNacimiento>
  </musico>
  <musico>
    <nombre>Johann Sebastian Bach</nombre>
    <apodo>El viejo peluca</apodo>
    <fechaNacimiento>21 de marzo de 1685</fechaNacimiento>
  </musico>
</musicos>

ERROR: el elemento <musico> no puede tener los dos elementos <nombre> y <apodo> al mismo tiempo.

Corregido: eliminar uno de los elementos o poner entre paréntesis el elemento eliminado en el elemento que no se ha eliminado.


<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE musicos [
  <!ELEMENT musicos (musico*)>
  <!ELEMENT musico ((nombre | apodo), fechaNacimiento)>
  <!ELEMENT nombre (#PCDATA)>
  <!ELEMENT apodo (#PCDATA)>
  <!ELEMENT fechaNacimiento (#PCDATA)>
]>

<musicos>
  <musico>
    <nombre>Antonio Vivaldi (El cura pelirrojillo)</nombre>
    <fechaNacimiento>4 de marzo de 1678</fechaNacimiento>
  </musico>
  <musico>
    <apodo>El viejo peluca (Johann Sebastian Bach)</apodo>
    <fechaNacimiento>21 de marzo de 1685</fechaNacimiento>
  </musico>
</musicos>


DTD - Ejercicio 1-8 - Teléfonos de emergencia
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE agenda [
  <!ELEMENT agenda (contacto*)>
  <!ELEMENT contacto (nombre, telefonoFijo*, telefonoMovil+)>
  <!ELEMENT nombre (#PCDATA)>
  <!ELEMENT telefonoFijo (#PCDATA)>
  <!ELEMENT telefonoMovil (#PCDATA)>
]>

<agenda>
  <contacto>
    <nombre>Ayuntamiento</nombre>
    <telefonoFijo>010</telefonoFijo>
  </contacto>
  <contacto>
    <nombre>Emergencias</nombre>
    <telefonoFijo>112 (Unión Europea)</telefonoFijo>
    <telefonoMovil>Desconocido</telefonoMovil>
    <telefonoFijo>911 (Estados Unidos)</telefonoFijo>
  </contacto>
</agenda>


ERROR: el segundo elemento <contacto> tiene los elementos <telefenoFijo> y <telefonoMovil> desordenados 
       y el primer elemento <contacto> debe tener un elemento <telefonoMovil>

Corregido: reordenar los elementos <telefenoFijo> y <telefonoMovil> en el segundo <contacto>
           y añadir un elemento <telefonoMovil> en el primer <contacto>


<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE agenda [
  <!ELEMENT agenda (contacto*)>
  <!ELEMENT contacto (nombre, telefonoFijo*, telefonoMovil+)>
  <!ELEMENT nombre (#PCDATA)>
  <!ELEMENT telefonoFijo (#PCDATA)>
  <!ELEMENT telefonoMovil (#PCDATA)>
]>

<agenda>
  <contacto>
    <nombre>Ayuntamiento</nombre>
    <telefonoFijo>010</telefonoFijo>
    <telefonoMovil></telefonoMovil>
  </contacto>
  <contacto>
    <nombre>Emergencias</nombre>
    <telefonoFijo>112 (Unión Europea)</telefonoFijo>
    <telefonoFijo>911 (Estados Unidos)</telefonoFijo>
    <telefonoMovil>Desconocido</telefonoMovil>
  </contacto>
</agenda>


DTD - Ejercicio 1-9 - El sistema solar
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE sistemaSolar [
  <!ELEMENT sistemaSolar (cuerpo*)>
  <!ELEMENT cuerpo (planeta|satelite|asteroide)>
  <!ELEMENT planeta (#PCDATA)>
  <!ELEMENT satelite (#PCDATA)>
  <!ELEMENT asteroide (#PCDATA)>
]>

<sistemaSolar>
  <cuerpo>
    <planeta>Tierra</planeta>
    <satelite>Luna</satelite>
  </cuerpo>
  <asteroide>Ceres</asteroide>
</sistemaSolar>


ERROR: el elemento <sistemaSolar> solo puede tener elementos <cuerpo>, no elementos <asteroide>
       y el elemento <cuerpo> solo puede tener un elemento (planeta, satelite o asteroide>

Corregido: crear elementos <cuerpo> para los elementos <planeta>, <satelite> y <asteroide>

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE sistemaSolar [
  <!ELEMENT sistemaSolar (cuerpo*)>
  <!ELEMENT cuerpo (planeta|satelite|asteroide)>
  <!ELEMENT planeta (#PCDATA)>
  <!ELEMENT satelite (#PCDATA)>
  <!ELEMENT asteroide (#PCDATA)>
]>

<sistemaSolar>
  <cuerpo>
    <planeta>Tierra</planeta>
  </cuerpo>
  <cuerpo>
  <satelite>Luna</satelite>
  </cuerpo>
  <cuerpo>
  <asteroide>Ceres</asteroide>
  </cuerpo>
</sistemaSolar>
